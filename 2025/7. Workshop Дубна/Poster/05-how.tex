\section{Квази-замороженная опция}
%------------------------------------------------
\begin{frame}
	\frametitle{Квази-замороженная опция}
	\par Для накопления ЭДМ сигнала, необходимо использование фильтра Вина.
	И соленоида для кольца в форме 8 на каждом прямолинейном участке.
	\newline
	\par Кольцо будет использовано в качестве аксионной антенны и могут быть исследованы аксионоподобные частицы низких масс.
	
	\begin{equation}
		\Delta\delta_{eq}=\frac{\gamma_s^2}{\gamma_s^2\alpha_0-1}\left[\frac{\delta_0^2}{2}\left(\alpha_1+\frac{3}{2}\frac{\beta_s^2}{\gamma_s^2}-\frac{\alpha_0}{\gamma_s^2}+\frac{1}{\gamma_s^4}\right)+\frac{\pi}{2L}\left(\epsilon_x\nu_x+\epsilon_y\nu_y\right)\right]
	\end{equation}
	
	\par Необходимость использования 3-х независимых семейств секступолей
	
\end{frame}
