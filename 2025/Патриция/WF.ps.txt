import numpy as np
import matplotlib.pyplot as plt

# Создаем массив значений y
y = np.linspace(1.0, 1.3, 100) # 100 точек для плавности

# Задаем G
G1 = 1.79
G2 = -0.14

# Вычисляем f(y) для каждого G по упрощенной формуле
f_y1 = 1 - (G1 + 1) / (G1 * y**2)
f_y2 = 1 - (G2 + 1) / (G2 * y**2)

# Настраиваем график
plt.figure(figsize=(10, 6))

# Строим первый график (G = 1.79)
plt.plot(y, f_y1, label=f'G = {G1}', color='blue', linewidth=2)
# Строим второй график (G = -0.14)
plt.plot(y, f_y2, label=f'G = {G2}', color='red', linewidth=2)

# Добавляем легенду, labels и сетку
plt.legend()
plt.xlabel('y')
plt.ylabel('f(y)')
plt.title('График функции f(y) = 1 - (G + 1) / (G * y²)')
plt.grid(True, linestyle='--', alpha=0.7)

# Устанавливаем пределы осей, чтобы лучше увидеть оба графика
plt.ylim(-2, 10) # Подбирается вручную для отображения обеих кривых

# Отображаем график
plt.show()
